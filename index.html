Caffyne Shell apt repo

Ubuntu 24.04 (noble)

echo "deb [signed-by=/etc/apt/keyrings/caffyne.gpg] https://caffyne-org.github.io/caffyne-deb noble main" \
  | sudo tee /etc/apt/sources.list.d/caffyne.list

Debian 13 (trixie)

echo "deb [signed-by=/etc/apt/keyrings/caffyne.gpg] https://caffyne-org.github.io/caffyne-deb trixie main" \
  | sudo tee /etc/apt/sources.list.d/caffyne.list
